/* TALERA FREE: local archive + recovery. No networking or second database. */
(function(){
'use strict';if(window.TaleraFreeBackup)return;
const DB='talera-lifeline-dev-local-v1',MAGIC='TALERA-FREE/1\n',MONTH=30*86400000;
const FIRST='talera-backup-first-use-v1',SEEN='talera-backup-reminder-v1',CONFIRMED='talera-backup-confirmed-v1';
const te=new TextEncoder(),td=new TextDecoder('utf-8',{fatal:true});
let prepared=null,pending=null,checking=false;
const LABEL_KEYS=["create","restore","save","saved","choose","preview","confirmRestore","ready","notVerified","empty","conflict","restored","lastConfirmed","neverConfirmed","reminderTitle","reminderMessage","later","already","preparing","checking"];
function L(n){return window.TaleraI18n?.t?.('backup.'+LABEL_KEYS[n])||'Back-up';}
function req(r){return new Promise((ok,bad)=>{r.onsuccess=()=>ok(r.result);r.onerror=()=>bad(r.error||Error('Opslagfout'))})}
function done(tx){return new Promise((ok,bad)=>{tx.oncomplete=()=>ok();tx.onabort=()=>bad(tx.error||Error('Transactie afgebroken'));tx.onerror=()=>bad(tx.error||Error('Transactie mislukt'))})}
function openDB(){return new Promise((ok,bad)=>{const r=indexedDB.open(DB,1);r.onupgradeneeded=()=>{for(const [s,key] of [['stories','id'],['files','key']])if(!r.result.objectStoreNames.contains(s))r.result.createObjectStore(s,{keyPath:key})};r.onsuccess=()=>ok(r.result);r.onerror=()=>bad(r.error)})}
async function current(){const db=await openDB();try{const tx=db.transaction(['stories','files'],'readonly');const [stories,files]=await Promise.all([req(tx.objectStore('stories').getAll()),req(tx.objectStore('files').getAll())]);return {stories,files}}finally{db.close()}}
async function digest(buffer){if(!crypto.subtle)throw Error('Bestandscontrole niet ondersteund');return [...new Uint8Array(await crypto.subtle.digest('SHA-256',buffer))].map(x=>x.toString(16).padStart(2,'0')).join('')}
function storiesValid(stories){if(!Array.isArray(stories)||stories.length>200000)throw Error('Ongeldige herinneringslijst');const ids=new Set();for(const s of stories){if(!s||typeof s.id!=='string'||s.id.length>192||!s.id||ids.has(s.id)||!Array.isArray(s.items||[]))throw Error('Ongeldig herinneringsrecord');ids.add(s.id)}}
async function prepare(){
 const {stories,files}=await current();storiesValid(stories);if(!stories.length)throw Error(L(9));if(files.length>200000)throw Error('Te veel bestanden');
 const desc=[],blobs=[],keys=new Set();let bytes=0;
 for(const f of files){if(!f||typeof f.key!=='string'||f.key.length>240||keys.has(f.key)||!(f.blob instanceof Blob))throw Error('Ongeldig mediabestand');keys.add(f.key);bytes+=f.blob.size;if(bytes>2147483648)throw Error('Back-up is te groot voor deze browser');desc.push({key:f.key,size:f.blob.size,mime:f.blob.type,sha256:await digest(await f.blob.arrayBuffer())});blobs.push(f.blob)}
 for(const s of stories)for(const it of s.items||[])if(!keys.has(it.key))throw Error('Ontbrekend origineel: '+it.key);
 const json=te.encode(JSON.stringify({type:'TALERA_FREE_BACKUP',version:1,created:new Date().toISOString(),stories,files:desc}));if(json.length>20000000)throw Error('Inhoudsoverzicht te groot');
 const prefix=MAGIC+String(json.length).padStart(12,'0')+'\n'+await digest(json)+'\n';
 const blob=new Blob([te.encode(prefix),json,...blobs],{type:'application/octet-stream'});
 return prepared={blob,filename:'TALERA-Free-'+new Date().toISOString().slice(0,10)+'.talerabackup',stories:stories.length,files:files.length};
}
async function read(file){
 if(!(file instanceof Blob)||file.size<100||file.size>2170000000)throw Error('Ongeldig back-upbestand');
 const prefix=te.encode(MAGIC).length,head=td.decode(await file.slice(0,prefix+78).arrayBuffer());
 if(!head.startsWith(MAGIC))throw Error('Verkeerd bestandsformaat');
 const n=head.slice(prefix,prefix+12),sha=head.slice(prefix+13,prefix+77);if(!/^\d{12}$/.test(n)||head[prefix+12]!=='\n'||!/^[a-f0-9]{64}$/.test(sha)||head[prefix+77]!=='\n')throw Error('Beschadigde kop');
 const len=Number(n),start=prefix+78;if(len>20000000||file.size<start+len)throw Error('Onvolledige reservekopie');
 const metadata=await file.slice(start,start+len).arrayBuffer();if(await digest(metadata)!==sha)throw Error('Beschadigd inhoudsoverzicht');
 const m=JSON.parse(td.decode(metadata));if(m.type!=='TALERA_FREE_BACKUP'||m.version!==1)throw Error('Onbekende back-upversie');
 storiesValid(m.stories);if(!Array.isArray(m.files)||m.files.length>200000)throw Error('Ongeldige media');
 const keys=new Set(),media=[];let offset=start+len;
 for(const f of m.files){if(!f||typeof f.key!=='string'||f.key.length>240||keys.has(f.key)||!Number.isSafeInteger(f.size)||f.size<0||f.size>2147483648||!/^[a-f0-9]{64}$/.test(f.sha256)||typeof f.mime!=='string')throw Error('Ongeldig bestandsoverzicht');keys.add(f.key);media.push({...f,offset});offset+=f.size}
 if(offset!==file.size)throw Error('Onvolledig of afwijkend mediabestand');
 for(const s of m.stories)for(const item of s.items||[])if(!keys.has(item.key))throw Error('Er ontbreekt een bestand van een herinnering');
 for(const f of media){const blob=file.slice(f.offset,f.offset+f.size);if(await digest(await blob.arrayBuffer())!==f.sha256)throw Error('Beschadigd mediabestand '+f.key)}
 return {file,m,media};
}
async function preview(file){
 const a=await read(file),live=await current(),stories=new Map(live.stories.map(x=>[x.id,x])),files=new Map(live.files.map(x=>[x.key,x])),newIds=[],newKeys=[],conflicts=[];
 for(const s of a.m.stories){const old=stories.get(s.id);if(!old)newIds.push(s.id);else if(JSON.stringify(old)!==JSON.stringify(s))conflicts.push('herinnering '+s.id)}
 for(const f of a.media){const old=files.get(f.key);if(!old)newKeys.push(f.key);else if(!(old.blob instanceof Blob)||old.blob.size!==f.size||await digest(await old.blob.arrayBuffer())!==f.sha256)conflicts.push('bestand '+f.key)}
 return pending={...a,newIds,newKeys,conflicts};
}
async function restore(info){
 if(info!==pending||info.conflicts.length)throw Error(L(10));
 const again=await preview(info.file);if(again.conflicts.length)throw Error(L(10));
 const ids=new Set(again.newIds),keys=new Set(again.newKeys),db=await openDB();
 try{const tx=db.transaction(['stories','files'],'readwrite'),complete=done(tx);
 for(const s of again.m.stories)if(ids.has(s.id))tx.objectStore('stories').add(s);
 for(const f of again.media)if(keys.has(f.key))tx.objectStore('files').add({key:f.key,blob:again.file.slice(f.offset,f.offset+f.size,f.mime)});
 await complete;pending=null;return {added:ids.size,existing:again.m.stories.length-ids.size};}finally{db.close()}
}
function node(tag,txt){const e=document.createElement(tag);e.textContent=txt;return e}
function btn(txt,fn){const b=node('button',txt);b.type='button';b.className='secondary';b.style.cssText='width:100%;min-height:44px;margin:7px 0;padding:10px 12px;border-radius:12px';b.addEventListener('click',fn);return b}
function confirmed(){try{const n=Number(localStorage.getItem(CONFIRMED));return n?new Date(n).toLocaleDateString(window.TaleraI18n?.locale||'nl-NL'):L(13)}catch{return L(13)}}
function acknowledge(){try{localStorage.setItem(CONFIRMED,String(Date.now()))}catch{}const e=document.querySelector('#talera-backup-last');if(e)e.textContent=L(12)+': '+confirmed()}
function download(){if(!prepared)return;const u=URL.createObjectURL(prepared.blob),a=document.createElement('a');a.href=u;a.download=prepared.filename;document.body.append(a);a.click();a.remove();setTimeout(()=>URL.revokeObjectURL(u),60000)}
async function save(){if(!prepared)return;const file=new File([prepared.blob],prepared.filename,{type:'application/octet-stream'});if(navigator.canShare?.({files:[file]})&&navigator.share){try{await navigator.share({files:[file],title:'TALERA Free'});return}catch(e){if(e.name==='AbortError')return}}download()}
function render(host){
 if(!host)return;host.querySelector('#talera-free-backup')?.remove();const root=document.createElement('section');root.id='talera-free-backup';root.style.cssText='padding:14px 0;border-top:1px solid #ccd8dd;margin-top:16px';
 const desc=node('p',window.TaleraI18n?.t?.('backup.intro')||L(7));root.append(desc);
 const last=node('p',L(12)+': '+confirmed());last.id='talera-backup-last';root.append(last);
 const status=node('p','');status.setAttribute('role','status');
 const output=document.createElement('div'),restoreZone=document.createElement('div');
 const create=btn(L(0),async()=>{create.disabled=true;status.textContent=L(18);try{await prepare();status.textContent=L(7)+' ('+prepared.stories+' herinneringen, '+prepared.files+' bestanden.)';output.replaceChildren(btn(L(2),()=>{void save().then(()=>status.textContent=L(8)).catch(e=>status.textContent=e.message)}),btn(L(3),()=>{acknowledge();status.textContent=L(8)}))}catch(e){status.textContent=e.message}finally{create.disabled=false}});
 const picker=document.createElement('input');picker.type='file';picker.accept='.talerabackup';picker.hidden=true;picker.setAttribute('aria-label',L(4));
 const choose=btn(L(1),()=>picker.click());
 picker.addEventListener('change',async()=>{restoreZone.replaceChildren();const file=picker.files?.[0];if(!file)return;status.textContent=L(19);try{const info=await preview(file);status.textContent='';restoreZone.append(node('p',L(5)+': '+window.TaleraI18n.t('backup.previewCounts').replace('%new%',info.newIds.length).replace('%existing%',info.m.stories.length-info.newIds.length).replace('%files%',info.newKeys.length)));if(info.conflicts.length){restoreZone.append(node('p',L(10)));return}const confirm=btn(L(6),async()=>{confirm.disabled=true;status.textContent=L(19);try{await restore(info);status.textContent=L(11);restoreZone.replaceChildren()}catch(e){status.textContent=e.message}finally{confirm.disabled=false}});restoreZone.append(confirm)}catch(e){status.textContent=e.message}});
 root.append(create,output,choose,picker,restoreZone,status);host.append(root);
}
async function remind(shell){
 if(checking)return;checking=true;try{const rows=await window.TaleraMemories?.list?.();if(!rows?.length)return;
 const now=Date.now(),first=Number(localStorage.getItem(FIRST)||0);if(!first){localStorage.setItem(FIRST,String(now));return}
 const seen=Number(localStorage.getItem(SEEN)||first);if(now-seen<MONTH)return;localStorage.setItem(SEEN,String(now));
 const p=document.createElement('div');p.id='talera-backup-reminder';p.style.cssText='position:absolute;z-index:9;bottom:12px;left:12px;right:12px;padding:16px;border-radius:16px;background:#fff;border:1px solid #ddd;box-shadow:0 2px 20px #0003;color:#153b51';
 p.append(node('strong',L(14)),node('p',L(15)));
 const close=()=>p.remove();p.append(btn(L(0),()=>{close();shell.querySelector('[data-view=settings]')?.click();setTimeout(()=>shell.querySelector('[data-view=storage]')?.click(),75)}),btn(L(17),()=>{acknowledge();close()}),btn(L(16),close));shell.querySelector('#talera-backup-reminder')?.remove();shell.append(p);
 }catch{}finally{checking=false}
}
window.TaleraFreeBackup={prepare,preview,restore,render,remind,acknowledge};
})();
