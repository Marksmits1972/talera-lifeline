'use strict';if(window.TaleraFreeBackup)return;
const DB='talera-lifeline-dev-local-v1',MAGIC='TALERA-FREE/1\n',MONTH=30*86400000,COMMENT_PREFIX='talera-thread-archive-v1:';
const FIRST='talera-backup-first-use-v1',SEEN='talera-backup-reminder-v1',CONFIRMED='talera-backup-confirmed-v1';
const te=new TextEncoder(),td=new TextDecoder('utf-8',{fatal:true});
let prepared=null,pending=null,checking=false;
const LABEL_KEYS=["create","restore","save","saved","choose","preview","confirmRestore","ready","notVerified","empty","conflict","restored","lastConfirmed","neverConfirmed","reminderTitle","reminderMessage","later","already","preparing","checking"];
function L(n){return window.TaleraI18n?.t?.('backup.'+LABEL_KEYS[n])||'Back-up';}
function req(r){return new Promise((ok,bad)=>{r.onsuccess=()=>ok(r.result);r.onerror=()=>bad(r.error||Error('Opslagfout'))})}
function done(tx){return new Promise((ok,bad)=>{tx.oncomplete=()=>ok();tx.onabort=()=>bad(tx.error||Error('Transactie afgebroken'));tx.onerror=()=>bad(tx.error||Error('Transactie mislukt'))})}

async function current(){const db=await openDB();try{const tx=db.transaction(['stories','files'],'readonly');const [stories,files]=await Promise.all([req(tx.objectStore('stories').getAll()),req(tx.objectStore('files').getAll())]);return {stories,files}}finally{db.close()}}
async function digest(buffer){if(!crypto.subtle)throw Error('Bestandscontrole niet ondersteund');return [...new Uint8Array(await crypto.subtle.digest('SHA-256',buffer))].map(x=>x.toString(16).padStart(2,'0')).join('')}
function storiesValid(stories){if(!Array.isArray(stories)||stories.length>200000)throw Error('Ongeldige herinneringslijst');const ids=new Set();for(const s of stories){if(!s||typeof s.id!=='string'||s.id.length>192||!s.id||ids.has(s.id)||!Array.isArray(s.items||[]))throw Error('Ongeldig herinneringsrecord');ids.add(s.id)}}

/* One merged conversation per memory, regardless of account or WhatsApp entry.
   Archive messages are inert, read-only local history; never POST them again. */
function ckey(localId,id){return COMMENT_PREFIX+encodeURIComponent(localId)+':'+encodeURIComponent(id)}
function cleanComment(c){
 if(!c||typeof c.id!=='string'||!c.id||c.id.length>128||typeof c.body!=='string'||c.body.length>5000||
    typeof c.display_name!=='string'||c.display_name.length>160||
    !Number.isFinite(Number(c.created_at))||Number(c.created_at)<0)throw Error('Ongeldige historische reactie');
 return {id:c.id,display_name:c.display_name,body:c.body,created_at:Number(c.created_at),
 parent_id:typeof c.parent_id==='string'?c.parent_id.slice(0,128):null,guest:!!c.guest};
}
function unionThreads(saved,online){
 const map=new Map();
 for(const c of [...saved,...online]){
  const n=cleanComment(c),prev=map.get(n.id);
  if(prev&&(prev.body!==n.body||prev.display_name!==n.display_name||prev.created_at!==n.created_at))
    throw Error('Verschillende inhoud voor dezelfde reactie: '+n.id);
  map.set(n.id,n);
 }
 return [...map.values()].sort((a,b)=>a.created_at-b.created_at||a.id.localeCompare(b.id));
}
async function localThreads(files){
 const map=new Map();
 for(const f of files){
  if(!f.key?.startsWith(COMMENT_PREFIX))continue;
  if(!(f.blob instanceof Blob)||f.blob.size>10000)throw Error('Beschadigde lokale reactie');
  const row=JSON.parse(await f.blob.text());
  if(typeof row.localId!=='string'||!row.localId||row.localId.length>192)throw Error('Ongeldige lokale koppeling');
  const c=cleanComment(row.comment),expected=ckey(row.localId,c.id);
  if(f.key!==expected)throw Error('Reactiekoppeling is niet geldig');
  if(!map.has(row.localId))map.set(row.localId,[]);
  map.get(row.localId).push(c);
 }
 return map;
}
async function onlineThread(localId){
 let link;
 try{link=await fetch('/api/pwa/memory-link?localMemoryId='+encodeURIComponent(localId),
  {credentials:'same-origin',cache:'no-store',headers:{accept:'application/json'}})}
 catch{return {status:'unverified',items:[]}}
 if(link.status===401)return {status:'not-authenticated',items:[]};
 if(!link.ok)return {status:'unverified',items:[]};
 const linked=await link.json();
 if(!linked.memoryId)return {status:'not-shared',items:[]};
 if(!/^[0-9a-f-]{36}$/i.test(linked.memoryId))throw Error('Ongeldige online koppeling');
 let response;
 try{response=await fetch('/api/threads/'+encodeURIComponent(linked.memoryId),
  {credentials:'same-origin',cache:'no-store',headers:{accept:'application/json'}})}
 catch{return {status:'unverified',items:[]}}
 if(!response.ok)return {status:'unverified',items:[]};
 const data=await response.json().catch(()=>null);
 if(!Array.isArray(data?.items))return {status:'unverified',items:[]};
 const items=data.items;
 // The existing thread API is currently capped at 200. Never silently claim
 // a complete archive if that boundary was reached.
 if(items.length>=200)throw Error('Dit gesprek bevat 200 of meer reacties. Volledige back-up kan nog niet worden gegarandeerd.');
 return {status:'verified',items:items.map(cleanComment)};
}
async function collectThreads(stories,files){
 const saved=await localThreads(files),threads=[],unverified=[];
 for(const story of stories){
  const previous=saved.get(story.id)||[],live=await onlineThread(story.id);
  if(live.status!=='verified'&&live.status!=='not-shared')unverified.push(story.id);
  const items=unionThreads(previous,live.items);
  if(items.length)threads.push({localId:story.id,items});
 }
 return {threads,unverified};
}
function validateThreads(threads,stories){
 if(!Array.isArray(threads)||threads.length>stories.length)throw Error('Ongeldig gesprekkenoverzicht');
 const ids=new Set(stories.map(s=>s.id)),seen=new Set();
 for(const thread of threads){
  if(!thread||typeof thread.localId!=='string'||!ids.has(thread.localId)||seen.has(thread.localId)||
    !Array.isArray(thread.items)||thread.items.length>200000)throw Error('Ongeldige herinneringskoppeling voor reacties');
  seen.add(thread.localId);
  if(unionThreads([],thread.items).length!==thread.items.length)throw Error('Dubbele reactienummers');
 }
}
async function archivedComments(localId){
 if(!localId||typeof localId!=='string')return [];
 const db=await openDB();
 try{
  const records=await req(db.transaction('files','readonly').objectStore('files').getAll());
  const map=await localThreads(records.filter(f=>String(f.key).startsWith(COMMENT_PREFIX)));
  return unionThreads(map.get(localId)||[],[]);
 }finally{db.close()}
}
function combineComments(archived,online){
 const list=new Map();
 for(const c of archived)list.set(c.id,{id:c.id,from:c.display_name,text:c.body,createdAt:c.created_at,parentId:c.parent_id});
 for(const c of (online?.comments||[])){
  if(!c?.id)continue;const prev=list.get(c.id);
  list.set(c.id,{...prev,...c});
 }
 return [...list.values()].sort((a,b)=>Number(a.createdAt||0)-Number(b.createdAt||0)||String(a.id).localeCompare(String(b.id)));
}
async function socialSnapshot(localId,live=null){
 const history=await archivedComments(localId);
 const comments=combineComments(history,live);
 return {hearted:!!live?.hearted,heartCount:Number(live?.heartCount)||0,
  comments,commentCount:comments.length,archived:history.length>0};
}

async function prepare(){
 const {stories,files}=await current();storiesValid(stories);if(!stories.length)throw Error(L(9));if(files.length>200000)throw Error('Te veel bestanden');
 const {threads,unverified}=await collectThreads(stories,files);
 const desc=[],blobs=[],keys=new Set();let bytes=0;
 for(const f of files.filter(f=>!String(f.key).startsWith(COMMENT_PREFIX))){if(!f||typeof f.key!=='string'||f.key.length>240||keys.has(f.key)||!(f.blob instanceof Blob))throw Error('Ongeldig mediabestand');keys.add(f.key);bytes+=f.blob.size;if(bytes>2147483648)throw Error('Back-up is te groot voor deze browser');desc.push({key:f.key,size:f.blob.size,mime:f.blob.type,sha256:await digest(await f.blob.arrayBuffer())});blobs.push(f.blob)}
 for(const s of stories)for(const it of s.items||[])if(!keys.has(it.key))throw Error('Ontbrekend origineel: '+it.key);
 const json=te.encode(JSON.stringify({type:'TALERA_FREE_BACKUP',version:2,created:new Date().toISOString(),stories,files:desc,threads}));if(json.length>20000000)throw Error('Inhoudsoverzicht te groot');
 const prefix=MAGIC+String(json.length).padStart(12,'0')+'\n'+await digest(json)+'\n';
 const blob=new Blob([te.encode(prefix),json,...blobs],{type:'application/octet-stream'});
 return prepared={blob,filename:'TALERA-Free-'+new Date().toISOString().slice(0,10)+'.talerabackup',stories:stories.length,files:desc.length,threads:threads.reduce((n,t)=>n+t.items.length,0),unverified};
}
async function read(file){
 if(!(file instanceof Blob)||file.size<100||file.size>2170000000)throw Error('Ongeldig back-upbestand');
 const prefix=te.encode(MAGIC).length,head=td.decode(await file.slice(0,prefix+78).arrayBuffer());
 if(!head.startsWith(MAGIC))throw Error('Verkeerd bestandsformaat');
 const n=head.slice(prefix,prefix+12),sha=head.slice(prefix+13,prefix+77);if(!/^\d{12}$/.test(n)||head[prefix+12]!=='\n'||!/^[a-f0-9]{64}$/.test(sha)||head[prefix+77]!=='\n')throw Error('Beschadigde kop');
 const len=Number(n),start=prefix+78;if(len>20000000||file.size<start+len)throw Error('Onvolledige reservekopie');
 const metadata=await file.slice(start,start+len).arrayBuffer();if(await digest(metadata)!==sha)throw Error('Beschadigd inhoudsoverzicht');
 const m=JSON.parse(td.decode(metadata));if(m.type!=='TALERA_FREE_BACKUP'||![1,2].includes(m.version))throw Error('Onbekende back-upversie');
 if(m.version===1)m.threads=[];validateThreads(m.threads,m.stories);
 storiesValid(m.stories);if(!Array.isArray(m.files)||m.files.length>200000)throw Error('Ongeldige media');
 const keys=new Set(),media=[];let offset=start+len;
 for(const f of m.files){if(!f||typeof f.key!=='string'||f.key.length>240||keys.has(f.key)||!Number.isSafeInteger(f.size)||f.size<0||f.size>2147483648||!/^[a-f0-9]{64}$/.test(f.sha256)||typeof f.mime!=='string')throw Error('Ongeldig bestandsoverzicht');keys.add(f.key);media.push({...f,offset});offset+=f.size}
 if(offset!==file.size)throw Error('Onvolledig of afwijkend mediabestand');
 for(const s of m.stories)for(const item of s.items||[])if(!keys.has(item.key))throw Error('Er ontbreekt een bestand van een herinnering');
 for(const f of media){const blob=file.slice(f.offset,f.offset+f.size);if(await digest(await blob.arrayBuffer())!==f.sha256)throw Error('Beschadigd mediabestand '+f.key)}
 return {file,m,media};
}
async function preview(file){
 const a=await read(file),live=await current(),stories=new Map(live.stories.map(x=>[x.id,x])),files=new Map(live.files.map(x=>[x.key,x])),newIds=[],newKeys=[],newComments=[],conflicts=[];
 for(const s of a.m.stories){const old=stories.get(s.id);if(!old)newIds.push(s.id);else if(JSON.stringify(old)!==JSON.stringify(s))conflicts.push('herinnering '+s.id)}
 for(const f of a.media){const old=files.get(f.key);if(!old)newKeys.push(f.key);else if(!(old.blob instanceof Blob)||old.blob.size!==f.size||await digest(await old.blob.arrayBuffer())!==f.sha256)conflicts.push('bestand '+f.key)}
 for(const thread of a.m.threads)for(const c of thread.items){
  const key=ckey(thread.localId,c.id),old=files.get(key);
  if(!old)newComments.push({key,localId:thread.localId,comment:c});
  else{
   if(!(old.blob instanceof Blob))conflicts.push('reactie '+c.id);
   else{
    let stored;try{stored=JSON.parse(await old.blob.text())}catch{}
    if(!stored||stored.localId!==thread.localId||JSON.stringify(cleanComment(stored.comment))!==JSON.stringify(cleanComment(c)))
      conflicts.push('reactie '+c.id);
   }
  }
 }
 return pending={...a,newIds,newKeys,newComments,conflicts};
}
async function restore(info){
 if(info!==pending||info.conflicts.length)throw Error(L(10));
 const again=await preview(info.file);if(again.conflicts.length)throw Error(L(10));
 const ids=new Set(again.newIds),keys=new Set(again.newKeys),db=await openDB();
 try{const tx=db.transaction(['stories','files'],'readwrite'),complete=done(tx);
 for(const s of again.m.stories)if(ids.has(s.id))tx.objectStore('stories').add(s);
 for(const f of again.media)if(keys.has(f.key))tx.objectStore('files').add({key:f.key,blob:again.file.slice(f.offset,f.offset+f.size,f.mime)});
 for(const c of again.newComments)tx.objectStore('files').add({key:c.key,blob:new Blob([JSON.stringify({localId:c.localId,comment:c.comment})],{type:'application/json'})});
 await complete;pending=null;return {added:ids.size,existing:again.m.stories.length-ids.size,comments:again.newComments.length};}finally{db.close()}
}
