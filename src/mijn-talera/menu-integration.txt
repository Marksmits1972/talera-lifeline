async function resumePremiumCloudMediaSync(){
 if(premiumSyncUi.working||navigator.onLine===false||!state.account?.id)return;
 try{if(localStorage.getItem(PREMIUM_DEVICE_OWNER_KEY)!==state.account.id)return}catch{return}
 const expectedAccountId=state.account.id;
 const ownerMatches=()=>state.account?.id===expectedAccountId&&localStorage.getItem(PREMIUM_DEVICE_OWNER_KEY)===expectedAccountId;
 const common={indexedDB:window.indexedDB,crypto:window.crypto,fetcher:(url,options)=>fetch(url,options),
   online:()=>navigator.onLine!==false,maxItems:10,expectedAccountId,ownerMatches};
 premiumSyncUi.working=true;premiumSyncUi.error='';
 let continueSoon=false;
 try{
  // Only small sequential batches, with account and entitlement checks per item.
  const [media,records]=await Promise.all([
   import('/__clean/premium-sync-coordinator.mjs'),
   import('/__clean/premium-records-coordinator.mjs')
  ]);
  const a=await media.runPremiumMediaSync(common);
  premiumSyncUi.phase=a.phase;premiumSyncUi.uploaded+=a.uploaded||0;premiumSyncUi.pending=a.pending||0;
  if(!ownerMatches())return;
  const b=await records.runPremiumRecordSync(common);
  premiumSyncUi.recordsPhase=b.phase;premiumSyncUi.recordsUploaded+=b.uploaded||0;premiumSyncUi.recordsPending=b.pending||0;
  continueSoon=(a.phase==='uploading'||b.phase==='uploading')&&ownerMatches();
  if(state.open&&state.view==='subscription')render();
 }catch(e){
  premiumSyncUi.error='Synchronisatie onderbroken. Bestaande Free-herinneringen blijven behouden; probeer opnieuw wanneer de verbinding werkt.';
  if(state.open&&state.view==='subscription')render();
 }finally{
  premiumSyncUi.working=false;
  if(continueSoon&&ownerMatches())setTimeout(()=>void resumePremiumCloudMediaSync(),1200);
 }
}
